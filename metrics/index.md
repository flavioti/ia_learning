
$$
\begin{aligned}
\text{Precision} &= \frac{\text{TP}}{\text{TP} + \text{FP}} \\[1.5em]
\text{Recall} &= \frac{\text{TP}}{\text{TP} + \text{FN}} \\[1.5em]
\text{F}_1\text{-Score} &= 2 \times \frac{\text{Precision} \times \text{Recall}}{\text{Precision} + \text{Recall}}
\end{aligned}
$$
